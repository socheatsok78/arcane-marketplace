NetBird Reverse Proxy for Arcane

edgenetbirdnetworking by socheatsok78
Raw 
x-arcane:
  icon: https://cdn.jsdelivr.net/gh/selfhst/icons@main/svg/netbird.svg
  urls:
    - https://netbird.io/
    - https://github.com/netbirdio/netbird

services:
  reverse-proxy:
    image: netbirdio/reverse-proxy:latest
    restart: unless-stopped
    ports:
      - "80:80"
      - "443:443/tcp"
      - "443:443/udp"
    environment:
      NB_PROXY_CERTIFICATE_DIRECTORY: /certs
      NB_PROXY_ALLOW_INSECURE: "true"
      NB_PROXY_MANAGEMENT_ADDRESS: "${NB_PROXY_MANAGEMENT_ADDRESS:-https://api.netbird.io}"
      NB_PROXY_ACME_CERTIFICATES: "true"
      NB_PROXY_DOMAIN: "${NB_PROXY_DOMAIN:?required}"
      NB_PROXY_LOG_LEVEL: ${NB_PROXY_LOG_LEVEL:-info}
      NB_PROXY_TOKEN: "${NB_PROXY_TOKEN:?required}"
      NB_PROXY_PRIVATE: "${NB_PROXY_PRIVATE:-true}"
      NB_PROXY_ADDRESS: ":443"
    volumes:
      - proxy_certs:/certs
volumes:
  proxy_certs:

# NB_PROXY_MANAGEMENT_ADDRESS=https://api.netbird.io

NB_PROXY_DOMAIN=change-this-to-your-netbird-proxy-domain
NB_PROXY_TOKEN=change-this-to-your-netbird-proxy-token

NB_PROXY_PRIVATE=true
NB_PROXY_LOG_LEVEL=info
